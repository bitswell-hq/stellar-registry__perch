The evaluator is about \fact{rule-eval-lines} lines of dependency-free
\code{no\_std} Rust without \code{unsafe}, in a subset Aeneas translates: no
traits, closures or iterator adaptors, and every loop a \code{while} over one
slice. Charon reads the MIR of rustc \fact{rustc-mir-nightly}, and
Aeneas~\cite{aeneas} translates it into a generated
\fact{extracted-lean-lines}-line Lean file, which rule-core's CI regenerates and
diffs. CI pins all \fact{exported-theorems} exported theorems to Lean's three
standard axioms and checks a ledger of \fact{claims-total} claims
(\fact{claims-proven} proven, \fact{claims-tested} tested, \fact{claims-assumed}
assumed, \fact{claims-unproven} unproven). We say \emph{proved} only of
kernel-checked statements. Each concerns one object: the generic \emph{model}
with its Soroban chain vocabulary; the \emph{translation}; a two-event
\emph{account model} with no Rust counterpart; a \emph{host model}, tested
against the host; or Perch's \emph{canonical-encoding} package. A translated
function has three outcomes, $\mathsf{ok}$, $\mathsf{fail}$ (panic) and
$\mathsf{div}$, so proving $\mathsf{ok}$ on every input rules out panics and
divergence; Rust's \code{Result<\_, Refusal>} is the value inside $\mathsf{ok}$.

\begin{perchdefinition}{D1}{contexts and authorizations}\label{def:d1}
A context is $c ::= \mathsf{call}(C, f, \vec v) \mid \mathsf{create}(h)$. An
authorization is $a = (\vec c, \vec\jmath, S, \ell, g)$: one entry's contexts,
the calls and creations of its tree in preorder, root first; one rule index per
context; the claimed signers; the ledger; and the account's generation.
\leannames{\lean{Chain.Context}, \lean{Authorization}}
\end{perchdefinition}

\begin{perchdefinition}{D2}{rules and \textit{allows}}\label{def:d2}
A rule $r$ has a scope $\mathit{sc}_r$ (a contract and function list, false on
creations), conditions $\Phi_r$, signers $\Sigma_r$, a threshold $k_r$, and
optionally a cap $(\lambda_r, P_r)$, a last ledger $u_r$ and a generation $g_r$;
$\mathit{allows}(r, c, S) \iff \mathit{sc}_r(c) \wedge \bigwedge_{\varphi \in
\Phi_r} \varphi(c) \wedge k_r \le |\Sigma_r \cap S|$, counting distinct signers.
A rule set allows $c$ under $S$ when one of its rules does; $R' \sqsubseteq R$
when $R$ allows every $(c, S)$ that $R'$ allows.
\leannames{\lean{Rule.allows}, \lean{RuleSet.narrower}}
\end{perchdefinition}

\begin{perchdefinition}{D3}{\textit{authorize} and runs}\label{def:d3}
Given one record per rule, $\rho$, $\mathit{authorize}(R, \rho, a)$ refuses
unless $|\vec\jmath| = |\vec c|$ and $|\rho| = |R|$, and then unless (i)~for each
$i$, $r = R_{j_i}$ exists, is unexpired ($\ell \le u_r$), is current ($g_r = g$
when present) and $\mathit{allows}(r, c_i, S)$; and (ii)~every $s \in S$ is a
signer of a chosen rule. It then (iii)~charges, in context order, each context
whose chosen rule is capped to that rule's record, and returns the records or
the first refusal. A run applies authorizations in order; a refused one leaves
the records unchanged.
\leannames{\lean{RuleSet.authorize}, \lean{RuleSet.Accepts}}
\end{perchdefinition}

\begin{perchdefinition}{D4}{window and \textit{spent}}\label{def:d4}
For period $P$, a charge at ledger $t$ is in the window at $L$ when
$L \mathbin{\dot-} P < t \le L$. $\mathit{spent}_j(R, \rho, \vec a)$ lists the
ledger and amount of every positive charge the accepted authorizations of run
$\vec a$ make to rule $j$.
\leannames{\lean{Params.inWindow}, \lean{RuleSet.spent}}
\end{perchdefinition}
